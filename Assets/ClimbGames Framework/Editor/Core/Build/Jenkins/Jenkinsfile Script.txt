@Library('climbgames-library') _
import climbgames.PipelineConfig
import climbgames.IBuildProcess

def settings
def buildProcess

pipeline {
    agent any

    options {
        skipDefaultCheckout()
    }    

    stages {
        stage('Setup') {
            
            environment {
                //GIT_URL = "${scm.userRemoteConfigs[0].url}"
                GIT_URL = "#GIT_URL#"
            }

            steps {

                checkout scmGit(
                    branches: [[name: "${params.BRANCH_NAME}"]],
                    userRemoteConfigs: [[
                        url: "${env.GIT_URL}",
                        credentialsId: 'github-climbgames'
                    ]]
                )

                script {

                    def config = setupConfig(this)
                    
                    def settingsFactory = load("Jenkins/Scripts/CustomSettings.groovy") as Closure
                    settings = settingsFactory(config)
                    buildProcess = createBuildProcess(config)
                }
            }
        }
        
        stage('Build') {
            steps {
                
                script {
                    
                    buildProcess.build(settings)
                }
            }
        }

        stage ('Deploy') {
            steps {

                script {
                    
                    buildProcess.deploy(settings)
                }
            }
        }
    }
}
