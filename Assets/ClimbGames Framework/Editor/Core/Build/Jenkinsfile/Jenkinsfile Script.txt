def buildConfig
def platformBuild

pipeline {
    agent any

    options {
        skipDefaultCheckout()
    }    

    stages {
        stage('Setup') {
            
            environment {
                //GIT_URL = "${scm.userRemoteConfigs[0].url}"
                GIT_URL = "#GIT_URL#"
            }

            steps {

                checkout scmGit(
                    branches: [[name: "${params.BRANCH_NAME}"]],
                    userRemoteConfigs: [[
                        url: "${env.GIT_URL}",
                        credentialsId: 'github-climbgames'
                    ]]
                )

                script {

                    buildConfig = load 'Jenkins/Scripts/config.groovy'
                    platformBuild = buildConfig.setup()
                }
            }
        }
        
        stage('Build') {
            steps {
                
                script {
                    
                    platformBuild.start(buildConfig)
                }
            }
        }

        stage ('Deploy') {
            steps {

                script {
                    
                    platformBuild.deploy(buildConfig)
                }
            }
        }
    }
}
